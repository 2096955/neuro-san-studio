// 1. Extract structured data (✓ checkmarks)
const extractStructuredData = (message) => {
  const extracted = {};
  const patterns = {
    name: /✓\s*name\s*:\s*["']?([^"'\n]+)["']?/i,
    email: /✓\s*email\s*:\s*["']?([^"'\n]+)["']?/i,
    phone: /✓\s*phone\s*:\s*["']?([^"'\n]+)["']?/i,
    rego: /✓\s*rego\s*:\s*["']?([^"'\n]+)["']?/i
  };
  Object.keys(patterns).forEach(key => {
    const match = message.match(patterns[key]);
    if (match && match[1]) extracted[key] = match[1].trim();
  });
  return extracted;
};

// 2. Detect brand (VWI/Ford/BMW)
const detectBrand = (message, context = {}) => {
  const lower = message.toLowerCase();
  const vwiKeywords = ['vw', 'volkswagen', 'vwi', 'tiguan', 'golf', 'id.4'];
  const fordKeywords = ['ford', 'mustang', 'f-150', 'ranger', 'explorer'];
  const bmwKeywords = ['bmw', '3 series', '5 series', 'x1', 'x3', 'x5'];
  
  if (vwiKeywords.some(k => lower.includes(k))) return 'VWI';
  if (fordKeywords.some(k => lower.includes(k))) return 'FORD';
  if (bmwKeywords.some(k => lower.includes(k))) return 'BMW';
  return context.brand || 'VWI'; // Default to VWI
};

// 3. Brand-aware prompt (prevents cross-contamination)
const buildBrandAwarePrompt = (userMessage, context) => {
  const brand = context.brand || 'VWI';
  const extracted = extractStructuredData(userMessage);
  
  let prompt = `You are a ${brand} customer service agent. `;
  prompt += `IMPORTANT: Never mention ${brand === 'VWI' ? 'Ford or BMW' : brand === 'FORD' ? 'Volkswagen or BMW' : 'Volkswagen or Ford'}. `;
  
  if (extracted.name) prompt += `Customer: ${extracted.name}. `;
  if (extracted.email) prompt += `Email: ${extracted.email}. `;
  if (extracted.phone) prompt += `Phone: ${extracted.phone}. `;
  if (extracted.rego) prompt += `Registration: ${extracted.rego}. `;
  
  return {
    prompt: `${prompt}\n\nCustomer request: ${userMessage}`,
    context: { ...context, brand, customer: { ...context.customer, ...extracted } }
  };
};
