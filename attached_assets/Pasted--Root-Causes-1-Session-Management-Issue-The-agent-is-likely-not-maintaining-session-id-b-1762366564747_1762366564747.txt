// Enhanced extraction function for structured data with checkmarks
const extractStructuredData = (message) => {
  const extracted = {};
  
  // Pattern: ✓ name: "value" or ✓ name: value
  const patterns = {
    name: /✓\s*name\s*:\s*["']?([^"'\n]+)["']?/i,
    email: /✓\s*email\s*:\s*["']?([^"'\n]+)["']?/i,
    phone: /✓\s*phone\s*:\s*["']?([^"'\n]+)["']?/i,
    rego: /✓\s*rego\s*:\s*["']?([^"'\n]+)["']?/i,
    registration: /✓\s*registration\s*:\s*["']?([^"'\n]+)["']?/i,
    vehicle: /✓\s*vehicle\s*:\s*["']?([^"'\n]+)["']?/i,
    model: /✓\s*model\s*:\s*["']?([^"'\n]+)["']?/i
  };
  
  // Extract from structured format
  Object.keys(patterns).forEach(key => {
    const match = message.match(patterns[key]);
    if (match && match[1]) {
      extracted[key] = match[1].trim();
    }
  });
  
  // Also extract from natural language (fallback)
  if (!extracted.name) {
    const nameMatch = message.match(/(?:name is|I'm|I am)\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+)*)/i);
    if (nameMatch) extracted.name = nameMatch[1];
  }
  
  if (!extracted.email) {
    const emailMatch = message.match(/[\w.-]+@[\w.-]+\.\w+/);
    if (emailMatch) extracted.email = emailMatch[0];
  }
  
  if (!extracted.phone) {
    const phoneMatch = message.match(/04\d{8}|\+614\d{8}/);
    if (phoneMatch) extracted.phone = phoneMatch[0];
  }
  
  if (!extracted.rego && !extracted.registration) {
    const regoMatch = message.match(/\b([A-Z0-9]{6,7})\s+(?:is the rego|rego)/i);
    if (regoMatch) extracted.rego = regoMatch[1];
  }
  
  return extracted;
};
