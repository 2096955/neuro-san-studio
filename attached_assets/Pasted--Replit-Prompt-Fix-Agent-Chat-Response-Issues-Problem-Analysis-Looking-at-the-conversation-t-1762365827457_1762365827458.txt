## Issue 1: Agent Memory - Maintain Conversation Context

**Problem:** Agent asks for customer details, user provides them, then agent asks again.

**Fix Required:**
1. Maintain `session_id` across all messages in the same conversation
2. Store user-provided information in conversation context
3. When user provides details (name, email, phone, rego), extract and remember them
4. Don't ask for information that was already provided in the current conversation

**Implementation:**
```javascript

  const updatedContext = { ...conversationContext, ...extractedInfo };

  const response = await fetch('/execute', {
    method: 'POST',
    body: JSON.stringify({
      task_id: `chat-${Date.now()}`,
      prompt: userMessage,
      context: {
        session_id: sessionId,  // CRITICAL: Maintain session
        agent_id: currentAgentId,
        user_details: updatedContext  // Pass accumulated context
      }
    })
  });

  const data = await response.json();

  // Save session ID from first response
  if (!sessionId && data.context.session_id) {
    setSessionId(data.context.session_id);
  }

  // Update conversation context
  setConversationContext(updatedContext);
};

// Helper function to extract user details
const extractUserDetails = (message) => {
  const details = {};

  // Extract email
  const emailMatch = message.match(/[\w.-]+@[\w.-]+\.\w+/);
  if (emailMatch) details.email = emailMatch[0];

  // Extract phone (Australian format)
  const phoneMatch = message.match(/04\d{8}|\+614\d{8}/);
  if (phoneMatch) details.phone = phoneMatch[0];

  // Extract name (after "name is")
  const nameMatch = message.match(/(?:name is|I'm|I am)\s+([A-Z][a-z]+(?:\s+[A-Z][a-z]+)*)/i);
  if (nameMatch) details.name = nameMatch[1];

  // Extract rego
  const regoMatch = message.match(/\b([A-Z0-9]{6,7})\s+is the rego|rego is\s+([A-Z0-9]{6,7})/i);
  if (regoMatch) details.rego = regoMatch[1] || regoMatch[2];

  return details;
};
```

```javascript
  if (lower.includes("finance")) return "finance";

const extractModel = (msg) => {
  const models = ["tiguan", "golf", "id.4", "id.3", "passat", "arteon", "polo"];

};

const extractBudget = (msg) => {
  const match = msg.match(/\$?\s*(\d{1,3}(?:[,\s]\d{3})*)/);
  return match ? match[0] : null;
};

const extractTimeline = (msg) => {
  const lower = msg.toLowerCase();
  if (lower.includes("urgent") || lower.includes("asap")) return "urgent";
  if (lower.includes("month")) return "within_month";
  if (lower.includes("week")) return "within_week";
  return null;
};

const extractFeatures = (msg) => {

  if (lower.includes("suv")) features.push("suv");
  if (lower.includes("safety")) features.push("safety");
  if (lower.includes("technology")) features.push("technology");
  return features;
```

```javascript
const buildAgentContext = (agentName, message, conversationContext) => {
  const agentContextBuilders = {
    'customer-service': buildCustomerServiceContext,
    'sales-agent': buildSalesAgentContext,
    'service-scheduling': buildServiceSchedulingContext,
    'recall-information': buildRecallInfoContext,
    'warranty-claims': buildWarrantyClaimsContext,
    'quality-control': buildQualityControlContext,
    'factory-efficiency': buildFactoryEfficiencyContext,
    'operations-coordinator': buildOperationsCoordinatorContext,
    'production-planning': buildProductionPlanningContext,
    'technical-service': buildTechnicalServiceContext,
    'manufacturing-ops': buildManufacturingOpsContext,
    'supply-chain': buildSupplyChainContext,
    'logistics-coordinator': buildLogisticsCoordinatorContext,
    'engineering-support': buildEngineeringSupportContext,
    'supplier-relations': buildSupplierRelationsContext,
    'parts-inventory': buildPartsInventoryContext
  };
  
  const builder = agentContextBuilders[agentName.toLowerCase().replace(/\s+/g, '-')];
  
  if (builder) {
    return builder(message, conversationContext);
  }
  
  // Default context builder
  return {
    prompt: message,
    context: conversationContext
  };
};

// Usage in sendMessage
const sendMessage = async (userMessage, agentName) => {
  // Build agent-specific context
  const { prompt, context } = buildAgentContext(agentName, userMessage, conversationContext);
  
  // Send to Salesforce with enriched context
  const response = await fetch('/execute', {
    method: 'POST',
    body: JSON.stringify({
      task_id: `chat-${Date.now()}`,
      prompt: prompt,
      context: {
        session_id: sessionId,
        agent_id: currentAgentId,
        ...context  // Include all agent-specific context
      }
    })
  });
  
  // Update conversation context
  setConversationContext(context);
  
  return response;
};
```
