## Conversation Continuity

All agents support multi-turn conversations using the `session_id`:

```bash
# Start conversation
RESPONSE=$(curl -s -X POST "$SERVICE_URL/execute" \
  -H "Content-Type: application/json" \
  -d '{
    "task_id": "conv-001",
    "prompt": "Hello, I need help with my VW",
    "context": {
      "agent_id": "0XxfI0000003NCzSAM"
    }
  }')

# Extract session_id from response
SESSION_ID=$(echo $RESPONSE | python3 -c "import sys, json; print(json.load(sys.stdin)['context']['session_id'])")

# Continue conversation
curl -s -X POST "$SERVICE_URL/execute" \
  -H "Content-Type: application/json" \
  -d "{
    \"task_id\": \"conv-002\",
    \"prompt\": \"My car makes a noise when braking\",
    \"context\": {
      \"session_id\": \"$SESSION_ID\",
      \"agent_id\": \"0XxfI0000003NCzSAM\"
    }
  }" | python3 -m json.tool
```

## Python Integration Example

```python
import requests

SERVICE_URL = "https://salesforce-agentforce-534348290993.us-central1.run.app"

# VWI Sales Agent
def query_vwi_sales(prompt, session_id=None):
    context = {"agent_id": "0XxfI0000003NEbSAM"}
    if session_id:
        context["session_id"] = session_id
    
    response = requests.post(
        f"{SERVICE_URL}/execute",
        json={
            "task_id": "python-sales-001",
            "prompt": prompt,
            "context": context
        }
    )
    return response.json()

# VWI Service Agent

    context = {"agent_id": "0XxfI0000003NCzSAM"}

        context["session_id"] = session_id
    
    response = requests.post(
        f"{SERVICE_URL}/execute",
        json={
            "task_id": "python-service-001",
            "prompt": prompt,
            "context": context
        }
    )
    return response.json()

# Example usage

# Continue conversation
```
